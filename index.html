`; card.addEventListener("click", () => launchGame(game)); grid.appendChild(card); }); document.getElementById("game-search").addEventListener("input", (e) => { renderGames(e.target.value.trim()); }); frame.src = url; } goBtn.addEventListener("click", () => loadURL(input.value.trim())); input.addEventListener("keydown", (e) => { if (e.key === "Enter") loadURL(input.value.trim()); }); reloadBtn.addEventListener("click", () => { frame.src = frame.src; }); fsBtn.addEventListener("click", () => { frame.requestFullscreen(); }); // Bookmark shortcuts (YouTube, Spotify, DuckDuckGo, Bing) document.querySelectorAll(".b-mark").forEach((btn) => { btn.addEventListener("click", () => { loadURL(btn.getAttribute("data-url"));
